def clean_text(value, max_len=5000):
    if not isinstance(value, str):
        return ""
    return value.strip()[:max_len]

def valid_http_url(url):
    try:
        p = urlparse(url)
        return p.scheme in {"http", "https"} and bool(p.netloc)
    except Exception:
        return False

@app.after_request
def security_headers(response):
    response.headers["X-Content-Type-Options"] = "nosniff"
    response.headers["X-Frame-Options"] = "SAMEORIGIN"
    response.headers["Referrer-Policy"] = "strict-origin-when-cross-origin"
    response.headers["Permissions-Policy"] = "camera=(), microphone=(), geolocation=()"
    response.headers["Content-Security-Policy"] = (
        "default-src 'self'; "
        "style-src 'self' 'unsafe-inline'; "
        "script-src 'self' 'unsafe-inline'; "
        "frame-ancestors 'self';"
    )
    return response

@app.post("/api/translate")
def translate():
    data = request.get_json(silent=True) or {}
    text = clean_text(data.get("text"), 5000)
    target = clean_text(data.get("target"), 30)

    if not text:
        return jsonify({"error": "Please enter text to translate."}), 400

    if target not in LANGUAGES.values():
        return jsonify({"error": "Unsupported target language."}), 400

    if target not in SUPPORTED_BY_GOOGLE:
        return jsonify({
            "error": "This language is listed, but the selected translation provider does not currently support it."
        }), 400

    try:
        translated = GoogleTranslator(source="auto", target=target).translate(text)
        return jsonify({"translation": translated})
    except Exception as exc:
        app.logger.exception("Translation failed: %s", exc)
        return jsonify({
            "error": "Translation service is temporarily unavailable. Please try again later."
        }), 503

def prompt_engineer():
    data = request.get_json(silent=True) or {}
